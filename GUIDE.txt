  interface SkillCardProps {
    number: string
    tag: string
    tagColor: "green" | "gray" | "cyan"
    icon: React.ReactNode
    title: string
    keywords: string[]
    cta: string
    ctaLink: string
  }

  const colorConfig = {
    green: { text: "text-neon-green", border: "border-neon-green", glow: "#39ff14" },
    gray: { text: "text-muted-gray", border: "border-muted-gray", glow: "#9ca3af" },
    cyan: { text: "text-neon-cyan", border: "border-neon-cyan", glow: "#00ffff" },
  }

  const SkillCard = ({
    number,
    tag,
    tagColor,
    icon,
    title,
    keywords,
    cta,
    ctaLink,
  }: SkillCardProps) => {
    const colors = colorConfig[tagColor]

    return (
      <div className="bg-dark-800 border border-dark-700 rounded-xl p-6 flex flex-col gap-4 hover:border-dark-600 transition-all duration-300">
        <div className="flex items-center gap-2 mb-2">
          <span
            className={`text-sm font-bold px-3 py-1 rounded-md ${colors.text}`}
            style={{
              textShadow: `0 0 8px ${colors.glow}`,
              boxShadow: `0 0 12px ${colors.glow}30`,
            }}
          >
            {number} / {tag}
          </span>
        </div>

        <div className="h-16 flex items-center justify-center text-4xl">
          {icon}
        </div>

        <h3 className="text-white font-bold text-lg uppercase tracking-wide leading-tight">
          {title}
        </h3>

        <div className="flex flex-wrap gap-2">
          {keywords.map((kw) => (
            <span
              key={kw}
              className="text-muted-gray text-xs border border-dark-700 px-2 py-1 rounded"
            >
              {kw}
            </span>
          ))}
        </div>

        <a
          href={ctaLink}
          className={`${colors.text} hover:text-white transition-colors duration-200 text-sm mt-auto tracking-wider`}
        >
          {cta} &gt;
        </a>
      </div>
    )
  }

  export default SkillCard

┌────────────────────────────────────────────────────────────┐
│ Concepto React #6: PROPS + TIPADO TypeScript               │
│                                                           │
│ interface SkillCardProps { ... }  ->  define la forma      │
│ que tienen las props que recibe el componente.            │
│ TypeScript verifica que le pasés los datos correctos.     │
│                                                           │
│ const SkillCard = ({ number, tag, tagColor, ... }:        │
│   SkillCardProps) => { ... }                              │
│ -> destructuración de props: saca cada prop directamente   │
│   como variable en vez de escribir props.number cada vez  │
│                                                           │
│ tagColor: "green" | "gray" | "cyan"  ->  union type:      │
│   solo acepta esos 3 valores exactos                     │
│                                                           │
│ icon: React.ReactNode  ->  puede recibir cualquier cosa    │
│   que React pueda renderizar (JSX, strings, etc.)        │
│                                                           │
│ keywords: string[]  ->  array de strings                  │
│                                                           │
│ Concepto React #7: RENDERIZADO CONDICIONAL                │
│                                                           │
│ const colors = colorConfig[tagColor]  ->  según el valor  │
│ de tagColor, elige un set de colores distinto. Así el     │
│ mismo componente se ve verde, gris o cyan según la prop. │
│                                                           │
│ Concepto React #8: INLINE STYLES                          │
│                                                           │
│ style={{ textShadow: `...`, boxShadow: `...` }}          │
│ -> los estilos inline en React se pasan como objeto        │
│   JS con camelCase (textShadow en vez de text-shadow)    │
│   y strings entre backticks para usar variables.         │
│                                                           │
│ Concepto React #9: TEMPLATE LITERALS (``)                │
│                                                           │
│ className={`${colors.text} hover:text-white ...`}        │
│ -> las backticks ` permiten interpolar variables con      │
│   ${variable} dentro de un string.                       │

RESUMEN DE CONCEPTOS DE REACT QUE APRENDISTE
══════════════════════════════════════════════════════════════════

  #1  Componente funcional             -> Navbar, Hero, etc.
  #2  Renderizado de listas (.map())   -> links del navbar
  #3  className (vs class en HTML)     -> en todos lados
  #4  export default / import          -> pasar componentes entre archivos
  #5  JSX (mezclar HTML y JS)          -> { } para expresiones
  #6  Props + TypeScript interfaces    -> SkillCard
  #7  Renderizado condicional          -> colorConfig según tagColor
  #8  Inline styles ({...})            -> textShadow, boxShadow
  #9  Template literals (`${}`)        -> className dinámico
  #10 Componentes presentacionales     -> Icons.tsx (solo SVG)
  #11 Fragmentos (<></>)               -> agrupar sin div extra
  #12 Composición de componentes       -> App.tsx como orquestador
